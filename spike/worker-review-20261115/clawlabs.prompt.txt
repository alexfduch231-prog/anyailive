155:      * Http отдаёт уже готовые строки, но если в чанке оказалось несколько строк (или пришёл

158:     public static List<String> payloads(String chunk, SseParser parser) {
159:         List<String> out = new ArrayList<String>();
160:         if (chunk == null) {
161:             return out;
162:         }
163:         SseParser p = parser == null ? new SseParser() : parser;
164:         if (chunk.indexOf('\n') < 0) {
165:             String s = chunk.trim();
166:             if (s.isEmpty()) {
167:                 return out;
168:             }
169:             String payload = s.startsWith("data:") ? p.feed(s) : s;
170:             if (payload != null && payload.length() > 0) {
171:                 out.add(payload);
172:             }
173:             return out;
174:         }
175:         String[] lines = chunk.split("\n");
176:         for (int i = 0; i < lines.length; i++) {
177:             String line = lines[i].replace("\r", "");
178:             String payload = p.feed(line);
179:             if (payload != null && payload.length() > 0) {
180:                 out.add(payload);
181:             }
182:         }
183:         return out;
184:     }

4:  * Разбор потока text/event-stream по строкам.
5:  * Чистый Java: тестируется на обычной JVM, без Android.

7: public class SseParser {
8: 
9:     /** @return JSON-полезная нагрузка строки, либо null если строку нужно пропустить. */
10:     public String feed(String line) {
11:         if (line == null) {
12:             return null;
13:         }
14:         String s = line.trim();
15:         if (s.isEmpty()) {
16:             return null;
17:         }
18:         if (s.charAt(0) == ':') {
19:             return null;
20:         }
21:         if (!s.startsWith("data:")) {
22:             return null;
23:         }
24:         String payload = s.substring(5).trim();
25:         if (payload.isEmpty() || "[DONE]".equals(payload)) {
26:             return null;
27:         }
28:         return payload;
29:     }
30: 
31:     /** Строка-терминатор потока: {@code data: [DONE]}. */
32:     public boolean isDone(String line) {
33:         if (line == null) {
34:             return false;
35:         }
36:         String s = line.trim();
37:         if (!s.startsWith("data:")) {
38:             return false;
39:         }
40:         return "[DONE]".equals(s.substring(5).trim());
41:     }
42: }
