44:     /** Собрать WAV из куска буфера. */
45:     public static byte[] encode(byte[] pcm, int off, int len, int sampleRate, int channels) {
46:         int rate = sampleRate <= 0 ? 16000 : sampleRate;
47:         int ch = channels <= 0 ? 1 : channels;
48:         int dataLen = pcm == null ? 0 : Math.max(0, Math.min(len, pcm.length - off));
49:         byte[] out = new byte[HEADER_BYTES + dataLen];
50:         int byteRate = rate * ch * 2;
51:         putTag(out, 0, "RIFF");
52:         putInt(out, 4, 36 + dataLen);
53:         putTag(out, 8, "WAVE");
54:         putTag(out, 12, "fmt ");

56:         putShort(out, 20, 1);           // PCM
57:         putShort(out, 22, ch);

59:         putInt(out, 28, byteRate);
60:         putShort(out, 32, ch * 2);      // blockAlign
61:         putShort(out, 34, 16);          // bits per sample
62:         putTag(out, 36, "data");
63:         putInt(out, 40, dataLen);
64:         if (dataLen > 0) {
65:             System.arraycopy(pcm, off, out, HEADER_BYTES, dataLen);
66:         }
67:         return out;
68:     }

103:     /**
104:      * Разобрать заголовок: идём по чанкам, поэтому переживаем и лишние блоки (LIST, fact),
105:      * и «потоковые» data с размером 0xFFFFFFFF.
106:      *
107:      * @return null, если это не RIFF/WAVE
108:      */
109:     public static Info parseHeader(byte[] data) {
110:         if (data == null || data.length < 12) {
111:             return null;
112:         }
113:         if (!tag(data, 0, "RIFF") || !tag(data, 8, "WAVE")) {
114:             return null;
115:         }
116:         Info info = new Info();
117:         int p = 12;
118:         while (p + 8 <= data.length) {
119:             String id = string(data, p, 4);
120:             long size = u32(data, p + 4);
121:             int body = p + 8;
122:             if ("fmt ".equals(id) && body + 16 <= data.length) {
123:                 info.channels = (int) u16(data, body + 2);
124:                 info.sampleRate = (int) u32(data, body + 4);
125:                 info.bits = (int) u16(data, body + 14);
126:             } else if ("data".equals(id)) {
127:                 info.dataOffset = body;
128:                 long avail = data.length - body;
129:                 long take = size > 0 && size <= avail ? size : avail;
130:                 info.dataLength = (int) Math.max(0, take);
131:                 return info.sampleRate > 0 ? info : null;
132:             }
133:             long advance = size + (size & 1);
134:             if (size <= 0 || advance > data.length) {
135:                 break;
136:             }
137:             p = (int) (body + advance);
138:         }
139:         return info.sampleRate > 0 ? info : null;
140:     }
