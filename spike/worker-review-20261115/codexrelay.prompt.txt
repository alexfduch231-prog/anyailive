3: /**
4:  * Граница колбэков голосового каскада на экране чата.
5:  *
6:  * Экран переписки включает и глушит каскад сколько угодно раз, поэтому у каждой
7:  * сессии свой номер. Колбэк, пришедший от старой сессии — уже поставленный в
8:  * очередь UI-потока, синхронно вызванный из engine.stop() или пришедший после
9:  * позднего разрешения на микрофон — не должен ни писать в историю, ни гасить ту
10:  * сессию, что успела начаться вместо него.
11:  *
12:  * Проверка делается непосредственно перед эффектом, поэтому внутри
13:  * runOnUiThread её нужно перепроверять, а не полагаться на момент постановки.
14:  */
15: public final class ChatSessionCallbacks {
16: 
17:     /** Получатель финального текста реплики. */
18:     public interface TextSink {
19:         void persist(String text);
20:     }
21: 
22:     /** Номер последней выданной сессии. */
23:     private int issued;
24:     /** Номер текущей сессии; 0 — каскад выключен. */
25:     private int current;
26: 
27:     /** Открыть новую сессию и вернуть её номер. */
28:     public int begin() {
29:         issued++;
30:         current = issued;
31:         return current;
32:     }
33: 
34:     /** Закрыть сессию: все её последующие колбэки станут no-op. Идемпотентно. */
35:     public void stop() {
36:         current = 0;
37:     }
38: 
39:     /** Идёт ли ещё сессия с таким номером. */
40:     public boolean active(int token) {
41:         return token != 0 && token == current;
42:     }
43: 
44:     /** Выполнить эффект колбэка, если его сессия всё ещё текущая. */
45:     public void runIfActive(int token, Runnable effect) {
46:         if (active(token) && effect != null) {
47:             effect.run();
48:         }
49:     }
50: 
51:     /** Сохранить финальный текст реплики: пустой текст в историю не попадает. */
52:     public void deliverText(int token, String text, TextSink sink) {
53:         if (!active(token) || sink == null) {
54:             return;
55:         }
56:         String clean = text == null ? "" : text.trim();
57:         if (clean.length() == 0) {
58:             return;
59:         }
60:         sink.persist(clean);
61:     }
62: }
