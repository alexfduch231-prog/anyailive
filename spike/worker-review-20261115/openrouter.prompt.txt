19:  * при выключенном AEC дуплекс (одновременная речь и ответ) ненадёжен — ассистент слышит сам себя,
20:  * и такой режим нужно вести полудуплексом.

93:     public synchronized boolean start(Listener l) {
94:         if (running) {
95:             return true;
96:         }
97:         if (l == null) {
98:             return false;
99:         }
100:         listener = l;
101:         try {
102:             int min = AudioRecord.getMinBufferSize(sampleRate, AudioFormat.CHANNEL_IN_MONO,
103:                     AudioFormat.ENCODING_PCM_16BIT);
104:             if (min <= 0) {
105:                 min = sampleRate; // 500 мс на 16 кГц — разумный запас, если платформа молчит
106:             }
107:             bufferBytes = Math.max(min * 2, readBytes * 4);
108:             bufferBytes &= ~1;
109: 
110:             record = open(MediaRecorder.AudioSource.VOICE_COMMUNICATION);
111:             if (record == null) {
112:                 record = open(MediaRecorder.AudioSource.MIC);
113:             }
114:             if (record == null) {
115:                 notifyError("Не удалось открыть микрофон: устройство не поддерживает запись");
116:                 return false;
117:             }
118:             record.startRecording();
119:             if (record.getRecordingState() != AudioRecord.RECORDSTATE_RECORDING) {
120:                 notifyError("Микрофон не начал запись (возможно, занят другим приложением)");
121:                 releaseRecord();
122:                 return false;
123:             }
124:             if (aecWanted) {
125:                 enableEffects();
126:             }
127:         } catch (SecurityException e) {
128:             notifyError("Нет разрешения на запись с микрофона");
129:             releaseRecord();
130:             return false;
131:         } catch (RuntimeException e) {
132:             notifyError("Микрофон недоступен: " + e.getMessage());
133:             releaseRecord();
134:             return false;
135:         }

137:         running = true;
138:         thread = new Thread(new Runnable() {
139:             @Override
140:             public void run() {
141:                 loop();
142:             }
143:         }, "AudioIn");
144:         thread.setDaemon(true);
145:         thread.start();
146:         return true;
147:     }

277:     private void loop() {
278:         byte[] buf = new byte[readBytes];
279:         while (running) {
280:             AudioRecord r = record;
281:             if (r == null) {
282:                 break;
283:             }
284:             int n;
285:             try {
286:                 n = r.read(buf, 0, buf.length);
287:             } catch (SecurityException e) {
288:                 notifyError("Разрешение на микрофон отозвано");
289:                 break;
290:             } catch (RuntimeException e) {
291:                 n = -1;
292:             }
293:             if (n < 0) {
294:                 if (running) {
295:                     notifyError("Ошибка чтения с микрофона");
296:                 }
297:                 break;
298:             }
299:             if (n == 0) {
300:                 continue;
301:             }
302:             Listener l = listener;
303:             if (l != null) {
304:                 try {
305:                     l.onPcm(buf, n);
306:                 } catch (RuntimeException ignored) {
307:                     // сбой в обработчике не должен убивать захват
308:                 }
309:             }
310:         }
311:     }
