5: /** Explicit local engine selection must never silently fall back to another provider. */
6: public final class TtsEnginePolicy {
7:     private TtsEnginePolicy() {}
8:     public static String resolve(String saved, Collection<String> installed) {
9:         String selected = saved == null ? "" : saved.trim();
10:         if (selected.length() == 0) return null;
11:         if (installed != null && installed.contains(selected)) return selected;
12:         throw new IllegalArgumentException("Выбранный TTS-движок не установлен: " + selected);
13:     }
14:     public static boolean matches(String requested, String actual) {
15:         return requested == null || requested.equals(actual);
16:     }
17: }
