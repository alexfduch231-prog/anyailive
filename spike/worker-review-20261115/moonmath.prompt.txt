32:     /** JSON одного чата вместе с сообщениями. */
33:     public static String chatJson(ChatStore.Chat chat, List<ChatStore.Msg> messages) {
34:         StringBuilder arr = new StringBuilder("[");
35:         for (int i = 0; i < messages.size(); i++) {
36:             ChatStore.Msg m = messages.get(i);
37:             if (i > 0) {
38:                 arr.append(',');
39:             }
40:             arr.append(Json.write()
41:                     .put("role", m.role)
42:                     .put("kind", m.kind)
43:                     .put("content", m.content)
44:                     .put("audio", m.audioPath == null ? "" : m.audioPath)
45:                     .put("truncated", m.truncated)
46:                     .put("at", m.at)
47:                     .done());
48:         }
49:         arr.append(']');
50:         return Json.write()
51:                 .put("chat", chat.title == null ? "" : chat.title)
52:                 .put("model_id", chat.modelId)
53:                 .put("updated_at", chat.updatedAt)
54:                 .put("exported_at", System.currentTimeMillis())
55:                 .putRaw("messages", arr.toString())
56:                 .done();
57:     }
58: 
59:     /** JSON всех чатов: массив объектов того же вида. */
60:     public static String allJson(List<ChatStore.Chat> chats, ChatStore store) {
61:         StringBuilder arr = new StringBuilder("[");
62:         for (int i = 0; i < chats.size(); i++) {
63:             ChatStore.Chat c = chats.get(i);
64:             if (i > 0) {
65:                 arr.append(',');
66:             }
67:             arr.append(chatJson(c, store.messages(c.id)));
68:         }
69:         arr.append(']');
70:         return Json.write()
71:                 .put("app", "anyailive")
72:                 .put("exported_at", System.currentTimeMillis())
73:                 .putRaw("chats", arr.toString())
74:                 .done();
75:     }
76: 
77:     /** Открывает системный диалог «куда сохранить» и запоминает текст. */
78:     public static void beginSave(Activity a, String suggestedName, String json) {
79:         pending = json;
80:         Intent i = new Intent(Intent.ACTION_CREATE_DOCUMENT);
81:         i.addCategory(Intent.CATEGORY_OPENABLE);
82:         i.setType("application/json");
83:         i.putExtra(Intent.EXTRA_TITLE, suggestedName);
84:         try {
85:             a.startActivityForResult(i, REQ_EXPORT);
86:         } catch (RuntimeException e) {
87:             pending = null;
88:             Toast.makeText(a, R.string.export_failed, Toast.LENGTH_LONG).show();
89:         }
90:     }
91: 
92:     /** Дописывает запомненный текст в файл, выбранный пользователем. */
93:     public static void finishSave(Activity a, int requestCode, int resultCode, Intent data) {
94:         if (requestCode != REQ_EXPORT) {
95:             return;
96:         }
97:         String text = pending;
98:         pending = null;
99:         if (resultCode != Activity.RESULT_OK || data == null || data.getData() == null || text == null) {
100:             return;
101:         }
102:         Uri uri = data.getData();
103:         OutputStream os = null;
104:         try {
105:             os = a.getContentResolver().openOutputStream(uri);
106:             if (os == null) {
107:                 throw new IllegalStateException("no output stream");
108:             }
109:             os.write(text.getBytes("UTF-8"));
110:             os.flush();
111:             Toast.makeText(a, R.string.export_done, Toast.LENGTH_SHORT).show();
112:         } catch (Exception e) {
113:             Toast.makeText(a, R.string.export_failed, Toast.LENGTH_LONG).show();
114:         } finally {
115:             if (os != null) {
116:                 try {
117:                     os.close();
118:                 } catch (Exception ignored) {
119:                     // поток уже не нужен
120:                 }
121:             }
122:         }
123:     }
124: 
125:     /** Имя файла вида anyailive-chat-20261114-1200.json. */
126:     public static String fileName(String prefix) {
127:         SimpleDateFormat f = new SimpleDateFormat("yyyyMMdd-HHmm", Locale.US);
128:         return prefix + "-" + f.format(new Date()) + ".json";
129:     }
130: }
