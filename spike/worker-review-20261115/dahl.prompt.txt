334:     private void play(byte[] pcm) {
335:         if (pcm == null || pcm.length == 0) {
336:             return;
337:         }
338:         AudioOut o = out;
339:         if (o != null) {
340:             o.write(pcm, pcm.length);
341:         }
342:         ByteArrayOutputStream c = clip;
343:         if (c != null) {
344:             synchronized (c) {
345:                 if (c.size() + pcm.length <= 24000 * 2 * 120) {
346:                     c.write(pcm, 0, pcm.length); // At most two minutes per clip.
347:                 } else {
348:                     clip = null; // Never export a silently truncated recording.
349:                 }
350:             }
351:         }

386:     /** Сбрасывает накопленный PCM реплики в wav; null — писать нечего или нельзя. */
387:     private String saveClip() {
388:         ByteArrayOutputStream c = clip;
389:         if (c == null || !new SettingsStore(this).getBool(SettingsStore.KEEP_HISTORY, true)) {
390:             return null;
391:         }
392:         byte[] pcm;
393:         synchronized (c) {
394:             if (c.size() == 0) return null;
395:             pcm = c.toByteArray();
396:             c.reset();
397:         }
398:         try {
399:             File dir = new File(getFilesDir(), "audio/" + chat.id);
400:             if (!dir.exists() && !dir.mkdirs()) {
401:                 return null;
402:             }
403:             File file = new File(dir, System.currentTimeMillis() + ".wav");
404:             if (!Wav.writeFile(file, pcm, AudioOut.SAMPLE_RATE, 1)) {
405:                 return null;
406:             }
407:             return file.getAbsolutePath();
408:         } catch (RuntimeException e) {
409:             return null;
410:         }

3: /** Android-free boundary for callbacks owned by a single Live screen. */
4: final class LiveSessionCallbacks {
5:     interface AssistantTurn {
6:         void discardAudio();
7:         void persist(String text);
8:     }
9: 
10:     // Confined to the Activity UI thread, including execution of callback effects.
11:     private boolean active;
12:     private boolean closed;
13: 
14:     boolean start() {
15:         if (closed || active) return false;
16:         active = true;
17:         return true;
18:     }
19: 
20:     void stop() {
21:         active = false;
22:         closed = true;
23:     }
24: 
25:     void runIfActive(Runnable action) {
26:         if (active) action.run();
27:     }
28: 
29:     void finishAssistant(String text, AssistantTurn turn) {
30:         if (!active) return;
31:         String clean = text == null ? "" : text.trim();
32:         if (clean.length() == 0) turn.discardAudio();
33:         else turn.persist(clean);
34:     }
