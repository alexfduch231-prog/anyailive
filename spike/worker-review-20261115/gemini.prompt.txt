139:     private void onInitDone(int status) {
140:         if (released) {
141:             return;
142:         }
143:         if (status != TextToSpeech.SUCCESS || tts == null) {
144:             unavailable("Синтез речи недоступен на этом устройстве");

150:             if (r == TextToSpeech.LANG_MISSING_DATA || r == TextToSpeech.LANG_NOT_SUPPORTED) {

2: import java.util.ArrayDeque;
3: /** Main-thread bounded pre-initialization queue. */
4: public final class PendingSpeechQueue<T> {
5:     private final int limit;
6:     private final ArrayDeque<T> queue = new ArrayDeque<T>();
7:     public PendingSpeechQueue(int limit) { this.limit=limit; }
8:     public boolean offer(T item) {
9:         if (queue.size() >= limit) return false;
10:         queue.add(item);
11:         return true;
12:     }
13:     public T poll() { return queue.poll(); }
14:     public boolean isEmpty() { return queue.isEmpty(); }
15:     public void clear() { queue.clear(); }
16: }
