5: import android.security.keystore.KeyGenParameterSpec;
6: import android.security.keystore.KeyProperties;
7: import android.util.Base64;
8: 
9: import java.security.KeyStore;
10: 
11: import javax.crypto.Cipher;
12: import javax.crypto.KeyGenerator;
13: import javax.crypto.SecretKey;
14: import javax.crypto.spec.GCMParameterSpec;
15: 
16: /**
17:  * Ключи API живут только здесь: AES-GCM, ключ шифрования в Android Keystore (не покидает устройство).
18:  * В БД и в экспорт попадает лишь имя записи (key_ref).
19:  */
20: public class SecretStore {
21: 
22:     private static final String PREF = "secrets";
23:     private static final String ALIAS = "anyailive_master";
24:     private static final int IV_LEN = 12;
25:     private static final int TAG_BITS = 128;
26: 
27:     private final SharedPreferences prefs;
28: 
29:     public SecretStore(Context c) {
30:         prefs = c.getApplicationContext().getSharedPreferences(PREF, Context.MODE_PRIVATE);
31:     }
32: 
33:     private SecretKey key() throws Exception {
34:         KeyStore ks = KeyStore.getInstance("AndroidKeyStore");
35:         ks.load(null);
36:         if (ks.containsAlias(ALIAS)) {
37:             return (SecretKey) ks.getKey(ALIAS, null);
38:         }
39:         KeyGenerator kg = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore");
40:         kg.init(new KeyGenParameterSpec.Builder(ALIAS,
41:                 KeyProperties.PURPOSE_ENCRYPT | KeyProperties.PURPOSE_DECRYPT)
42:                 .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
43:                 .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
44:                 .setKeySize(256)
45:                 .build());
46:         return kg.generateKey();
47:     }
48: 
49:     /** @return true, если секрет сохранён. */
50:     public boolean put(String ref, String secret) {
51:         if (ref == null || ref.isEmpty() || secret == null) {
52:             return false;
53:         }
54:         try {
55:             Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
56:             c.init(Cipher.ENCRYPT_MODE, key());
57:             byte[] iv = c.getIV();
58:             byte[] ct = c.doFinal(secret.getBytes("UTF-8"));
59:             byte[] all = new byte[iv.length + ct.length];
60:             System.arraycopy(iv, 0, all, 0, iv.length);
61:             System.arraycopy(ct, 0, all, iv.length, ct.length);
62:             prefs.edit().putString(ref, Base64.encodeToString(all, Base64.NO_WRAP)).apply();
63:             return true;
64:         } catch (Exception e) {
65:             return false;
66:         }
67:     }
68: 
69:     /** @return секрет или null (нет записи либо ключ недоступен). Исключения не вылетают. */
70:     public String get(String ref) {
71:         if (ref == null || ref.isEmpty()) {
72:             return null;
73:         }
74:         String stored = prefs.getString(ref, null);
75:         if (stored == null) {
76:             return null;
77:         }
78:         try {
79:             byte[] all = Base64.decode(stored, Base64.NO_WRAP);
80:             byte[] iv = new byte[IV_LEN];
81:             byte[] ct = new byte[all.length - IV_LEN];
82:             System.arraycopy(all, 0, iv, 0, IV_LEN);
83:             System.arraycopy(all, IV_LEN, ct, 0, ct.length);
84:             Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
85:             c.init(Cipher.DECRYPT_MODE, key(), new GCMParameterSpec(TAG_BITS, iv));
86:             return new String(c.doFinal(ct), "UTF-8");
87:         } catch (Exception e) {
88:             return null;
89:         }
90:     }
91: 
92:     public void delete(String ref) {
93:         if (ref != null) {
94:             prefs.edit().remove(ref).apply();
95:         }
96:     }
97: }
