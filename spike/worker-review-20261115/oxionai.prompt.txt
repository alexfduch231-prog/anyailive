26:     public String get(String key, String def) {
27:         android.database.Cursor c = helper.getReadableDatabase().query(Db.T_SETTINGS,
28:                 new String[]{"value"}, "key=?", new String[]{key}, null, null, null);
29:         try {
30:             return c.moveToFirst() ? c.getString(0) : def;
31:         } finally {
32:             c.close();
33:         }
34:     }

37:         ContentValues v = new ContentValues();
38:         v.put("key", key);
39:         v.put("value", value == null ? "" : value);
40:         helper.getWritableDatabase().insertWithOnConflict(Db.T_SETTINGS, null, v,
41:                 android.database.sqlite.SQLiteDatabase.CONFLICT_REPLACE);
42:     }

10: public final class VoiceConfig {
11: 
12:     public static final String ENGINE_AUTO = "auto";
13:     public static final String ENGINE_LIVE = "live";
14:     public static final String ENGINE_CASCADE = "cascade";
15: 
16:     private VoiceConfig() {
17:     }
18: 
19:     /** Копия модели с подставленными моделями STT/TTS и голосом из настроек. */
20:     public static Model apply(SettingsStore settings, Model model) {
21:         return apply(model, settings.get(SettingsStore.STT_MODEL, ""),
22:                 settings.get(SettingsStore.TTS_MODEL, ""), settings.get(SettingsStore.VOICE, ""));
23:     }
24: 
25:     /** Model-specific choices win; global settings only fill missing values. */
26:     public static Model apply(Model model, String stt, String tts, String voice) {
27:         if (model == null) {
28:             return null;
29:         }
30:         Model m = model.copy();
31:         m.sttModel = pick(m.sttModel, stt);
32:         m.ttsModel = pick(m.ttsModel, tts);
33:         // A global OpenAI voice must not replace Gemini's provider-specific default.
34:         m.voice = pick(m.voice, Model.LIVE_GEMINI.equals(m.liveProtocol) ? "" : voice);
35:         return m;
36:     }
37: 
38:     /** Resolve the same policy before permission prompts and after permission is granted. */
39:     public static String route(String requested, Model model) {
40:         boolean live = model != null && (Model.LIVE_OPENAI.equals(model.liveProtocol)
41:                 || Model.LIVE_GEMINI.equals(model.liveProtocol));
42:         if (ENGINE_CASCADE.equals(requested)) {
43:             return ENGINE_CASCADE;
44:         }
45:         if (ENGINE_LIVE.equals(requested)) {
46:             return live ? ENGINE_LIVE : "unsupported";
47:         }
48:         return live ? ENGINE_LIVE : ENGINE_CASCADE;
49:     }
50: 
51:     /** Язык распознавания и синтеза; по умолчанию — то, что передали. */
52:     public static String language(SettingsStore settings, String fallback) {

54:     }
55: 
56:     public static String engine(SettingsStore settings) {
57:         String e = settings.get(SettingsStore.ENGINE, ENGINE_AUTO);
58:         return e == null || e.length() == 0 ? ENGINE_AUTO : e;
59:     }

61:     private static String pick(String preferred, String fallback) {
62:         if (preferred != null && preferred.trim().length() > 0) {
63:             return preferred.trim();

65:         return fallback == null ? "" : fallback.trim();
