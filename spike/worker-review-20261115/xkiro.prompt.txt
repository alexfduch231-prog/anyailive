416: 
417:     private void postFail(final String msg) {
418:         main.post(new Runnable() {
419:             @Override
420:             public void run() {
421:                 fail(msg);
422:             }
423:         });
424:     }
425: 
426:     private void fail(String msg) {
427:         Callback c = cb;
428:         releaseAll();
429:         if (c != null) {
430:             c.onError(msg);
431:         }
432:     }
433: 
434:     private void releaseAll() {
435:         running = false;
436:         state = IDLE;
437:         SystemStt s = stt;
438:         stt = null;
439:         if (s != null) {
440:             s.stop();
441:         }
442:         SystemTts t = tts;
443:         tts = null;
444:         if (t != null) {
445:             t.release();
446:         }
447:         AudioOut out = audioOut;
448:         audioOut = null;
449:         if (out != null) {
450:             out.stop();
451:         }
452:         synchronized (speech) {
453:             speech.clear();
454:         }
455:         ExecutorService w = chatWorker;
456:         chatWorker = null;
457:         if (w != null) {
458:             w.shutdownNow();
459:         }
460:         cb = null;
461:     }
462: 
463:     @Override
464:     public void stop() {
465:         Runnable r = new Runnable() {
466:             @Override
467:             public void run() {
468:                 releaseAll();
469:             }
470:         };
471:         if (Looper.myLooper() == Looper.getMainLooper()) {
472:             r.run();
473:         } else {
474:             main.post(r);
475:         }
476:     }

479:     public File cacheDir() {
