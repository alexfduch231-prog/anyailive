4: public final class Handler {
5:     public static final LinkedBlockingQueue<Runnable> queue = new LinkedBlockingQueue<Runnable>();
6:     public Handler(Looper looper) {}
7:     public boolean post(Runnable r) { queue.add(r); return true; }
8:     public boolean postDelayed(Runnable r, long delay) { return true; }
9:     public static Runnable next() throws Exception {
10:         Runnable r = queue.poll(5, TimeUnit.SECONDS);
11:         if (r == null) throw new AssertionError("worker did not post callback");
