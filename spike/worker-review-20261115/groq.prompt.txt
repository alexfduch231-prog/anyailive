172:         final int gen = generation;

181:                     return;

249:     private void speakPhrase(String phrase, int gen) {
250:         if (gen != generation || !running) return;
251:         if (TtsRoutingPolicy.useSystemTts(selectedTtsEngine, systemTtsFallback)) {
252:             speakSystem(phrase, gen);
253:             return; // Explicit local selection never reaches /audio/speech.
254:         }
255:         String body = HttpWire.ttsJson(ttsModel(), voice(), phrase);
256:         HttpWire.Bin r = HttpWire.post(HttpWire.url(model.baseUrl, "/audio/speech"),
257:                 HttpWire.headers(apiKey, model.headers), "application/json",
258:                 HttpWire.utf8(body), TTS_TIMEOUT_MS);
259:         if (gen != generation || !running) return;
260:         byte[] pcm = r.code >= 400 || r.code <= 0 ? null : binaryAudio(r.data);
261:         if (pcm == null || pcm.length == 0) {
262:             if (fallbackToSystemTts(r.code)) {
263:                 systemTtsFallback = true; // у модели нет /audio/speech — больше не пробуем
264:             }
265:             speakSystem(phrase, gen); // хоть какой-то звук лучше молчания
266:             return; // Main dispatch owns this phrase's inflight decrement.
267:         } else {
268:             AudioOut o = out;
269:             if (o != null) {
270:                 o.write(pcm, pcm.length);
271:             }
272:         }
273:         done(gen);
274:     }

286:     public static byte[] binaryAudio(byte[] data) {
287:         if (data == null || data.length == 0) {
288:             return null;
