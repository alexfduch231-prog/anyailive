8: /** Process-local timeline. Private entries are never replayed into persistent storage. */
9: final class MemoryHistory<T> {
10:     interface Storage<T> {
11:         List<T> load(long chatId);
12:         T save(long chatId, T message);
13:     }
14: 
15:     private final Map<Long, List<T>> timelines = new HashMap<Long, List<T>>();
16:     private long temporaryId = -1;
17: 
18:     synchronized T append(long chatId, T message, boolean keepHistory, Storage<T> storage) {
19:         List<T> timeline = timeline(chatId, storage);
20:         T added = keepHistory ? storage.save(chatId, message) : message;
21:         timeline.add(added);
22:         return added;
23:     }
24: 
25:     synchronized List<T> messages(long chatId, Storage<T> storage) {
26:         return new ArrayList<T>(timeline(chatId, storage));
27:     }
28: 
29:     synchronized List<T> lastMessages(long chatId, int count, Storage<T> storage) {
30:         List<T> all = timeline(chatId, storage);
31:         int from = Math.max(0, all.size() - Math.max(0, count));
32:         return new ArrayList<T>(all.subList(from, all.size()));
33:     }
34: 
35:     synchronized void remove(long chatId) {
36:         timelines.remove(chatId);
37:     }
38: 
39:     synchronized long nextTemporaryId() {
40:         return temporaryId--;
41:     }
42: 
43:     private List<T> timeline(long chatId, Storage<T> storage) {
44:         List<T> result = timelines.get(chatId);
45:         if (result == null) {
46:             result = new ArrayList<T>(storage.load(chatId));
47:             timelines.put(chatId, result);
48:         }
49:         return result;
50:     }
51: }
