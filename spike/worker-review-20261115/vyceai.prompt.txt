3: /**
4:  * Определение речи по энергии (RMS) для PCM16 LE.
5:  *
6:  * Логика:
7:  * - первые {@link #CALIBRATION_MS} мс считаются калибровкой: кадры не считаются речью,
8:  *   а оценка шума берётся как минимум энергии за этот отрезок (минимум устойчив к тому,
9:  *   что человек начал говорить сразу);
10:  * - порог = max(threshold, 3 × оценка шума), оценка шума подтягивается скользящим средним,
11:  *   пока речи нет;
12:  * - речь начинается с первого кадра выше порога (без задержки — нужно для перебивания)
13:  *   и заканчивается после silenceMs тишины.
14:  *
15:  * Класс чистый Java (без android.*) и полностью проверяется на JVM.
16:  */
17: public final class Vad {
18: 
19:     public static final int DEFAULT_THRESHOLD = 800;
20:     public static final int DEFAULT_SILENCE_MS = 700;
21:     public static final int CALIBRATION_MS = 300;
22: 
23:     private final int frameBytes;
24:     private final int sampleRate;
25:     private final int frameMs;
26: 
27:     private int threshold = DEFAULT_THRESHOLD;
28:     private int silenceMs = DEFAULT_SILENCE_MS;
29:     private int calibrationMs = CALIBRATION_MS;

142:     /**
143:      * Скормить очередной кусок PCM (может быть любой длины, кадры нарезаются внутри).
144:      *
145:      * @return true, если в куске был хотя бы один кадр речи
146:      */
147:     public boolean feed(byte[] pcm, int len) {
148:         if (pcm == null || len < 2) {
149:             return false;
150:         }
151:         int n = Math.min(len, pcm.length);
152:         boolean anySpeech = false;
153:         int off = 0;
154:         while (off + 2 <= n) {
155:             int take = Math.min(frameBytes, n - off);
156:             take &= ~1;
157:             if (take < 2) {
158:                 break;
159:             }
160:             if (frame(rms(pcm, off, take))) {
161:                 anySpeech = true;
162:             }
163:             off += take;
164:         }
165:         return anySpeech;
166:     }
167: 
168:     private boolean frame(int r) {
169:         lastRms = r;
170:         if (calibrating) {
171:             calibrationFrames++;
172:             noise = noise == 0 ? r : Math.min(noise, r);
173:             if (calibrationFrames * frameMs >= calibrationMs) {
174:                 calibrating = false;
175:             }
176:             return false;
177:         }
178:         boolean voice = r >= effectiveThreshold();
179:         if (voice) {
180:             silenceAcc = 0;
181:             if (!inSpeech) {
182:                 inSpeech = true;
183:                 startedFlag = true;
184:             }
185:         } else if (inSpeech) {
186:             silenceAcc += frameMs;
187:             if (silenceAcc >= silenceMs) {
188:                 inSpeech = false;
189:                 silenceAcc = 0;
190:                 endedFlag = true;
191:             }
192:         } else {
193:             silenceAcc = 0;
194:             noise = (noise * 7 + r) / 8;
195:         }
196:         return voice;
197:     }
198: 
199:     private static int rms(byte[] b, int off, int len) {
200:         int samples = len / 2;
201:         if (samples <= 0) {
202:             return 0;
203:         }
204:         long sum = 0;
205:         for (int i = 0; i < samples; i++) {
206:             int s = (short) ((b[off + i * 2] & 0xFF) | (b[off + i * 2 + 1] << 8));
207:             sum += (long) s * s;
208:         }
209:         double mean = (double) sum / samples;
210:         return (int) Math.sqrt(mean);
211:     }
