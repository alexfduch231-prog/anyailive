7: /** Asynchronous PCM16 playback. Flush and short nonblocking writes share one monitor. */
8: public final class AudioOut {
9:     public static final int SAMPLE_RATE = 24000;
10:     private final int sampleRate;
11:     private final int channels;
12:     private final Object lock = new Object();
13:     private final AudioOutBuffer buffer;
14:     // All mutable playback state is protected by lock, including AudioTrack access.
15:     private AudioTrack track;
16:     private Thread writer;
17:     private boolean running;
18:     private String lastError;
19: 
20:     public AudioOut() { this(SAMPLE_RATE, 1); }
21: 
22:     public AudioOut(int sampleRate, int channels) {
23:         this.sampleRate = sampleRate <= 0 ? SAMPLE_RATE : sampleRate;
24:         this.channels = channels == 2 ? 2 : 1;
25:         buffer = new AudioOutBuffer(this.sampleRate, this.channels);
26:     }
27: 
28:     public int sampleRate() { return sampleRate; }
29:     public int channels() { return channels; }
30:     public String lastError() { synchronized (lock) { return lastError; } }
31: 
32:     public boolean start() {
33:         synchronized (lock) {
34:             if (running) return true;
35:             releaseTrackLocked();
36:             buffer.flush();
37:             int mask = channels == 2 ? AudioFormat.CHANNEL_OUT_STEREO : AudioFormat.CHANNEL_OUT_MONO;
38:             try {
39:                 int min = AudioTrack.getMinBufferSize(sampleRate, mask, AudioFormat.ENCODING_PCM_16BIT);
40:                 if (min <= 0) min = sampleRate * channels;
41:                 track = new AudioTrack(AudioManager.STREAM_MUSIC, sampleRate, mask,
42:                         AudioFormat.ENCODING_PCM_16BIT,
43:                         Math.max(min * 2, sampleRate * channels / 2), AudioTrack.MODE_STREAM);
44:                 if (track.getState() != AudioTrack.STATE_INITIALIZED) {
45:                     lastError = "Динамик не поддерживает " + sampleRate + " Гц";
46:                     releaseTrackLocked();
47:                     return false;
48:                 }
49:                 track.play();
50:             } catch (RuntimeException e) {
51:                 lastError = "Не удалось открыть динамик: " + e.getMessage();
52:                 releaseTrackLocked();
53:                 return false;
54:             }
55:             lastError = null;
56:             running = true;
57:             writer = new Thread(new Runnable() {
58:                 @Override public void run() { loop(); }
59:             }, "AudioOut");
60:             writer.setDaemon(true);
61:             writer.start();
62:             return true;
63:         }
64:     }
65: 
66:     /** Copy and queue PCM without waiting for hardware buffer space. */
67:     public void write(byte[] pcm, int len) {
68:         if (pcm == null || len <= 0) return;
69:         int n = Math.min(len, pcm.length);
70:         n -= n % (2 * channels);
71:         if (n == 0) return;
72:         synchronized (lock) {
73:             if (!running || track == null) return;
74:             byte[] copy = new byte[n];
75:             System.arraycopy(pcm, 0, copy, 0, n);
76:             buffer.enqueue(copy);
77:             lock.notifyAll();
78:         }
79:     }
80: 
81:     /** Invalidate queued AND writer-owned audio before discarding the hardware buffer. */
82:     public void flush() {
83:         synchronized (lock) {
84:             buffer.flush();
85:             if (track != null) {
86:                 try {
87:                     track.pause();
88:                     track.flush(); // pause + flush resets AudioTrack's playback head to zero.
89:                     if (running) track.play();
90:                 } catch (RuntimeException e) {
91:                     lastError = "Не удалось сбросить динамик: " + e.getMessage();
92:                     running = false;
93:                     releaseTrackLocked();
94:                 }
95:             }
96:             lock.notifyAll();
97:         }
98:     }

118:                 return 0;

120:         }

124:     public int queued() { synchronized (lock) { return buffer.queued(); } }

126:     public void stop() {
127:         Thread oldWriter;
128:         synchronized (lock) {
129:             running = false;
130:             buffer.flush();
131:             oldWriter = writer;
132:             writer = null;
133:             releaseTrackLocked();
134:             lock.notifyAll();
135:         }
136:         // Never join while holding lock: the exiting writer needs it. A concurrent start
137:         // is safe because loop also checks the writer identity, not just running.
138:         if (oldWriter != null && oldWriter != Thread.currentThread()) {
139:             try {
140:                 oldWriter.join(500);
141:             } catch (InterruptedException e) {
142:                 Thread.currentThread().interrupt();
143:             }
144:         }
145:     }
146: 
147:     public void release() { stop(); }
148: 
149:     private void releaseTrackLocked() {
150:         AudioTrack t = track;
151:         track = null;
152:         if (t == null) return;
153:         try { t.pause(); } catch (RuntimeException ignored) { }
154:         try { t.flush(); } catch (RuntimeException ignored) { }
155:         try { t.stop(); } catch (RuntimeException ignored) { }
156:         try { t.release(); } catch (RuntimeException ignored) { }
157:     }
158: 
159:     private void loop() {
160:         int failures = 0;
161:         AudioOutBuffer.Chunk previous = null;
162:         try {
163:             for (;;) {
164:                 synchronized (lock) {
165:                     if (!running || writer != Thread.currentThread()) return;
166:                     AudioOutBuffer.Chunk chunk = buffer.current();
167:                     if (chunk == null) {
168:                         lock.wait();
169:                         continue;
170:                     }
171:                     if (chunk != previous) {
172:                         previous = chunk;
173:                         failures = 0;
174:                     }
175:                     if (!buffer.isCurrent(chunk)) continue;
176:                     // At most 10 ms per call, nonblocking even when AudioTrack is full.
177:                     // Checking the epoch, submitting bytes and accounting are atomic vs flush.
178:                     int count = Math.min(chunk.data.length - chunk.offset,
179:                             Math.max(1, sampleRate / 100) * channels * 2);
180:                     int written;
181:                     try {
182:                         written = track.write(chunk.data, chunk.offset, count, AudioTrack.WRITE_NON_BLOCKING);
183:                     } catch (RuntimeException e) {
184:                         written = -1;
185:                     }
186:                     if (written > 0) {
187:                         buffer.accept(chunk, written);
188:                         failures = 0;
189:                     } else {
190:                         if (++failures > 200) {
191:                             lastError = "Динамик перестал принимать звук";
192:                             running = false;
193:                             buffer.flush();
194:                             releaseTrackLocked();
195:                             return;
196:                         }
197:                         lock.wait(5); // releases monitor so flush/stop never wait for buffer space
198:                     }
199:                 }
200:                 Thread.yield();
201:             }
202:         } catch (InterruptedException e) {
203:             Thread.currentThread().interrupt();
204:         }
205:     }
206: }

3: import java.util.ArrayDeque;
4: 
5: /** Pure playback bookkeeping. Caller must hold AudioOut's monitor for every access. */
6: final class AudioOutBuffer {
7:     static final class Chunk {
8:         final byte[] data;
9:         final long epoch;
10:         int offset;
11:         Chunk(byte[] data, long epoch) { this.data = data; this.epoch = epoch; }
12:     }
13: 
14:     private final int sampleRate;
15:     private final int frameBytes;
16:     private final ArrayDeque<Chunk> queue = new ArrayDeque<Chunk>();
17:     private Chunk current;
18:     private long epoch;
19:     private long writtenBytes;
20:     private long previousHead;
21:     private long headWraps;
22: 
23:     AudioOutBuffer(int sampleRate, int channels) {
24:         this.sampleRate = sampleRate;
25:         this.frameBytes = channels * 2;
26:     }
27: 
28:     void enqueue(byte[] data) { queue.add(new Chunk(data, epoch)); }
29:     int queued() { return queue.size(); }
30:     Chunk current() {
31:         if (current == null) current = queue.poll();
32:         return current;
33:     }
34:     boolean hasPending() { return current != null || !queue.isEmpty(); }
35:     boolean isCurrent(Chunk chunk) {
36:         return chunk != null && chunk == current && chunk.epoch == epoch;
37:     }
38:     void accept(Chunk chunk, int count) {
39:         if (!isCurrent(chunk) || count <= 0) return;
40:         chunk.offset += count;
41:         writtenBytes += count;
42:         if (chunk.offset == chunk.data.length) current = null;
43:     }
44:     void flush() {
45:         epoch++;
46:         queue.clear();
47:         current = null;
48:         writtenBytes = 0;
49:         previousHead = 0;
50:         headWraps = 0;
51:     }
52:     long writtenFrames() { return writtenBytes / frameBytes; }
53:     long playedFrames(int head) {
54:         long unsigned = head & 0xffffffffL;
55:         if (unsigned < previousHead) headWraps++;
56:         previousHead = unsigned;
57:         return (headWraps << 32) + unsigned;
58:     }
59:     boolean hasAudio(int head) {
60:         long played = playedFrames(head);
61:         return hasPending() || played < writtenFrames();
62:     }
63:     long playedMillis(int head) {
64:         return Math.min(playedFrames(head), writtenFrames()) * 1000 / sampleRate;
65:     }
66: }
